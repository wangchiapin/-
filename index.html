name: 部署到 GitHub Pages（自動注入 Gemini 金鑰） on: push: branches: ["main"] # 如果你的預設分支不是 main（例如是 master），請把這裡改成你的分支名稱 workflow_dispatch: # 允許在 GitHub 網頁上手動點擊執行一次 # 讓這個 workflow 有權限發布到 GitHub Pages permissions: contents: read pages: write id-token: write concurrency: group: "pages" cancel-in-progress: false jobs: build-and-deploy: runs-on: ubuntu-latest environment: name: github-pages url: ${{ steps.deployment.outputs.page_url }} steps: - name: 取出程式碼 uses: actions/checkout@v4 - name: 準備要發布的資料夾（把整個 repo 內容複製一份，排除 .git 和 workflow 設定本身） run: | mkdir -p _site rsync -av --exclude='.git' --exclude='.github' --exclude='_site' ./ _site/ - name: 把佔位符換成真正的 Gemini API 金鑰 env: GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }} run: | # 把 _site 資料夾裡所有 .html 檔案中的 %%GEMINI_API_KEY%% 換成真正的金鑰 find _site -type f -name "*.html" -print0 | xargs -0 sed -i "s|%%GEMINI_API_KEY%%|${GEMINI_API_KEY}|g" - name: 設定 GitHub Pages uses: actions/configure-pages@v5 - name: 上傳成 Pages 要用的產物 uses: actions/upload-pages-artifact@v3 with: path: _site - name: 正式發布 id: deployment uses: actions/deploy-pages@v4
